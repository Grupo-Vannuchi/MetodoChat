**Objetivo:** cada pessoa entra no painel com o próprio e-mail e a própria senha, e a sessão fica gravada no
banco. O Sair, a troca de senha, a senha nova dada pelo administrador e a desativação derrubam o acesso na
hora. A senha única de hoje vira o login do Eduardo na primeira entrada, sem passar por ninguém. Nesta entrega
todo login é administrador; o login comum e as contas por login são a entrega 2.

**Arquitetura:**
- uma migração nova (a 022) cria `logins` e `sessoes`, que a conferência da partida passa a exigir;
- as regras puras moram em `lib/logins.ts`, e o banco em `lib/logins-repositorio.ts`, com a trava do último
  administrador por `for update`;
- `lib/sessao.ts` confere o cookie no banco, e os 18 lugares que conferiam a assinatura do cookie v2 passam
  por ela: o porteiro, o `exigirSessao` comum e as 6 cópias, as 9 rotas e o layout;
- a entrada (`app/entrar/`) usa `lib/entrada.ts`, com a passagem da senha única;
- duas páginas novas: Logins (`app/logins/`, só administrador) e Minha conta (`app/minha-conta/`, com o Sair
  como action).

**Stack:** a do projeto, no Next.js 16.4.3: Vitest (as três suítes), Testing Library no jsdom, o
`react-dom/server` para desenhar as páginas do servidor nos testes, o `scrypt` do `node:crypto` e o Postgres
no container para a integração.

import { createHash, randomBytes, scrypt, timingSafeEqual } from "node:crypto";

// AS REGRAS PURAS DOS LOGINS (docs/specs/2026-11-04-logins.md): o e-mail, o nome, a senha, o hash da senha e
// o código da sessão. Nada aqui toca o banco nem o cookie; quem faz isso é lib/logins-repositorio.ts e
// lib/sessao.ts.

/** A frase de toda entrada errada: o e-mail que não existe, a senha errada e o login desativado. */
export const FRASE_DA_ENTRADA_ERRADA = "E-mail ou senha incorretos.";

const EMAIL = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

/** O e-mail sem os espaços das pontas e em minúsculas, ou nulo se não for um e-mail de até 254 caracteres. */
export function normalizarEmail(bruto: string): string | null {
  const e = bruto.trim().toLowerCase();
  if (e.length > 254 || !EMAIL.test(e)) return null;
  return e;
}

/** O nome sem os espaços das pontas, de 1 a 60 caracteres, ou nulo. */
export function nomeValido(bruto: string): string | null {
  const n = bruto.trim();
  return n.length >= 1 && n.length <= 60 ? n : null;
}

/** A senha nova vai de 10 a 200 caracteres. */
export function senhaValida(senha: string): boolean {
  return senha.length >= 10 && senha.length <= 200;
}

// O SCRYPT (achado 172 da auditoria). O mínimo da OWASP é N=2^17, r=8, p=1; o padrão do Node é N=2^14.
// Medido em 04/11/2026 numa máquina de desenvolvimento: ~50 ms com 2^14 e ~480 ms com 2^17. A escolha
// final é a medida na Vercel, na prova (FASE do preview do plano). O N, o r e o p moram no próprio hash,
// então um hash feito com outros parâmetros continua conferindo, e dá para subir o N depois.
//
// O `maxmem` vai explícito: o scrypt usa 128 * N * r bytes, e com N=2^17 isso dá 128 MiB, acima dos
// 32 MiB do padrão do Node.
export const PARAMETROS_DO_SCRYPT = { N: 2 ** 17, r: 8, p: 1 } as const;
const TAMANHO_DO_HASH = 64;
const TAMANHO_DO_SAL = 16;

function scryptEmPromessa(senha: string, sal: Buffer, N: number, r: number, p: number): Promise<Buffer> {
  return new Promise((resolve, reject) => {
    scrypt(senha, sal, TAMANHO_DO_HASH, { N, r, p, maxmem: 256 * N * r }, (erro, chave) =>
      erro ? reject(erro) : resolve(chave)
    );
  });
}

/** O hash da senha, no formato `scrypt$N$r$p$<sal base64>$<hash base64>`. */
export async function hashDaSenha(senha: string): Promise<string> {
  const { N, r, p } = PARAMETROS_DO_SCRYPT;
  const sal = randomBytes(TAMANHO_DO_SAL);
  const hash = await scryptEmPromessa(senha, sal, N, r, p);
  return `scrypt$${N}$${r}$${p}$${sal.toString("base64")}$${hash.toString("base64")}`;
}

/** A senha confere com o hash guardado? Um hash torto responde que não, sem estourar. */
export async function senhaConfere(senha: string, guardado: string): Promise<boolean> {
  const partes = guardado.split("$");
  if (partes.length !== 6 || partes[0] !== "scrypt") return false;
  const [N, r, p] = partes.slice(1, 4).map(Number);
  // O TETO DOS PARÂMETROS: o hash vem do banco, e não de quem entra, mas um N ou um r enorme pediria
  // memória sem fim ao scrypt. 2^20 e 32 sobram para qualquer subida razoável.
  const potenciaDeDois = Number.isSafeInteger(N) && N > 1 && N <= 2 ** 20 && (N & (N - 1)) === 0;
  const rOk = Number.isSafeInteger(r) && r >= 1 && r <= 32;
  const pOk = Number.isSafeInteger(p) && p >= 1 && p <= 16;
  if (!potenciaDeDois || !rOk || !pOk) return false;
  const sal = Buffer.from(partes[4], "base64");
  const esperado = Buffer.from(partes[5], "base64");
  if (sal.length === 0 || esperado.length !== TAMANHO_DO_HASH) return false;
  try {
    const hash = await scryptEmPromessa(senha, sal, N, r, p);
    return timingSafeEqual(hash, esperado);
  } catch {
    return false;
  }
}

// O CÓDIGO DA SESSÃO: 32 bytes sorteados, em base64url, que vão no cookie. O banco guarda só o SHA-256 dele.
const CODIGO_DE_SESSAO = /^[A-Za-z0-9_-]{43}$/;

export function novoCodigoDeSessao(): string {
  return randomBytes(32).toString("base64url");
}

/** O formato do código. O cookie v2 de antes de 04/11/2026 (`v2.<expira>.<assinatura>`) não passa. */
export function codigoDeSessaoBemFormado(codigo: string | undefined): codigo is string {
  return typeof codigo === "string" && CODIGO_DE_SESSAO.test(codigo);
}

export function hashDoCodigoDeSessao(codigo: string): string {
  return createHash("sha256").update(codigo).digest("hex");
}

import { codigoDeSessaoBemFormado, hashDoCodigoDeSessao, novoCodigoDeSessao } from "./logins";

// resto deste projeto: o prazo da sessão nasce de `now()`, e a conferência pede `expira_em > now()`.

/** A sessão vale 30 dias, como o cookie assinado de antes. */

/** O login como a página Logins o lê. O `senha_hash` nunca sai daqui. */

  id: string;
  email: string;
  nome: string;
  administrador: boolean;
  criado_em: Date;
  desativado_em: Date | null;

  nome: string;

  email: string;
  nome: string;
  senhaHash: string | null;
}): Promise<{ id: string } | { erro: "email_repetido" }> {
  try {
    const [linha] = (await sql().query(
      `insert into logins (email, nome, senha_hash) values ($1, $2, $3) returning id`,
      [dados.email, dados.nome, dados.senhaHash]

  } catch (e) {
    const erro = e as { code?: string; constraint_name?: string };
    if (erro.code === "23505" && erro.constraint_name === "logins_email_key") return { erro: "email_repetido" };

  }

export async function loginParaEntrar(
  email: string
): Promise<{ id: string; senha_hash: string | null; desativado: boolean } | null> {
  const [linha] = (await sql().query(
    `select id, senha_hash, desativado_em is not null as desativado from logins where email = $1`,
    [email]
  )) as { id: string; senha_hash: string | null; desativado: boolean }[];

/**
 * A PASSAGEM DA SENHA ÚNICA (seção 6 da spec): grava o hash só num login ativo que ainda está sem senha. A
 * segunda entrada já não passa por aqui, e duas entradas ao mesmo tempo gravam uma vez só.
 */
export async function gravarSenhaDaPassagem(id: string, senhaHash: string): Promise<boolean> {
  const linhas = (await sql().query(
    `update logins set senha_hash = $2
      where id = $1 and senha_hash is null and desativado_em is null
      returning id`,
    [id, senhaHash]
  )) as { id: string }[];
  return linhas.length === 1;
}

export async function listarLogins(): Promise<Login[]> {
  return (await sql().query(
    `select id, email, nome, administrador, criado_em, desativado_em from logins order by criado_em asc, email asc`

/** O hash da senha de um login ativo, para a Minha conta conferir a senha atual. */
export async function senhaHashDoLogin(id: string): Promise<string | null> {

    [id]
  )) as { senha_hash: string | null }[];

}

/** A senha nova dada pelo administrador: grava e derruba TODAS as sessões daquele login. */
export async function definirSenha(id: string, senhaHash: string): Promise<boolean> {
  if (!UUID.test(id)) return false;
  return sql().begin(async (tx) => {
    const linhas = (await tx.query(`update logins set senha_hash = $2 where id = $1 returning id`, [
      id,
      senhaHash,
    ])) as { id: string }[];
    if (linhas.length !== 1) return false;
    await tx.query(`delete from sessoes where login_id = $1`, [id]);
    return true;
  });
}

/** A troca da própria senha: grava e derruba as OUTRAS sessões, mantendo a de quem trocou. */
export async function trocarPropriaSenha(id: string, senhaHash: string, sessaoQueFica: string): Promise<boolean> {

  return sql().begin(async (tx) => {
    const linhas = (await tx.query(
      `update logins set senha_hash = $2 where id = $1 and desativado_em is null returning id`,
      [id, senhaHash]
    )) as { id: string }[];
    if (linhas.length !== 1) return false;
    await tx.query(`delete from sessoes where login_id = $1 and id <> $2`, [id, sessaoQueFica]);
    return true;
  });

export type ResultadoDaDesativacao = "desativado" | "proprio" | "ultimo_administrador" | "nao_encontrado";

/**
 * DESATIVA UM LOGIN E DERRUBA AS SESSÕES DELE, com as duas travas da spec: ninguém desativa o próprio
 * login, e o último administrador ativo fica.
 *
 * A SEGUNDA TRAVA PRECISA DO `for update` (achado 168). Numa transação comum, em READ COMMITTED, dois
 * administradores que desativam um ao outro ao mesmo tempo contariam 2 ativos cada um, e o painel ficaria
 * com 0. Com as linhas dos administradores ativos travadas ANTES de contar, a segunda transação espera a
 * primeira e, ao seguir, relê o `where`, que já não pega a linha desativada.
 */
export async function desativarLogin(alvo: string, quemPede: string): Promise<ResultadoDaDesativacao> {
  if (!UUID.test(alvo)) return "nao_encontrado";
  if (alvo === quemPede) return "proprio";
  return sql().begin(async (tx) => {
    const ativos = (await tx.query(
      `select id from logins where administrador and desativado_em is null for update`
    )) as { id: string }[];
    const [linha] = (await tx.query(
      `select administrador, desativado_em is not null as desativado from logins where id = $1 for update`,
      [alvo]
    )) as { administrador: boolean; desativado: boolean }[];
    if (!linha || linha.desativado) return "nao_encontrado";
    if (linha.administrador && ativos.length <= 1) return "ultimo_administrador";
    await tx.query(`update logins set desativado_em = now() where id = $1`, [alvo]);
    await tx.query(`delete from sessoes where login_id = $1`, [alvo]);
    return "desativado";
  });
}

export async function reativarLogin(id: string): Promise<boolean> {
  if (!UUID.test(id)) return false;
  const linhas = (await sql().query(
    `update logins set desativado_em = null where id = $1 and desativado_em is not null returning id`,
    [id]
  )) as { id: string }[];
  return linhas.length === 1;
}

/** Abre uma sessão e devolve o código do cookie, que só existe nesta resposta. Limpa as vencidas. */
export async function criarSessao(loginId: string): Promise<string> {
  const codigo = novoCodigoDeSessao();
  await sql().query(
    `insert into sessoes (token_hash, login_id, expira_em) values ($1, $2, now() + make_interval(days => $3))`,
    [hashDoCodigoDeSessao(codigo), loginId, VALIDADE_DA_SESSAO_DIAS]
  );
  // Limpeza oportunista, como a do login_attempts: sem ela a tabela cresceria para sempre.
  await sql().query(`delete from sessoes where expira_em <= now()`);
  return codigo;
}

/** O login de uma sessão válida: não vencida, de login ativo. Código torto ou desconhecido dá null. */
export async function sessaoPorCodigo(codigo: string | undefined): Promise<LoginDaSessao | null> {
  if (!codigoDeSessaoBemFormado(codigo)) return null;
  const [linha] = (await sql().query(
    `select l.id, l.email, l.nome, l.administrador, s.id as sessao_id
       from sessoes s join logins l on l.id = s.login_id
      where s.token_hash = $1 and s.expira_em > now() and l.desativado_em is null`,
    [hashDoCodigoDeSessao(codigo)]
  )) as LoginDaSessao[];
  return linha ?? null;
}

/** O Sair: apaga a sessão no banco. */
export async function apagarSessao(codigo: string | undefined): Promise<void> {
  if (!codigoDeSessaoBemFormado(codigo)) return;
  await sql().query(`delete from sessoes where token_hash = $1`, [hashDoCodigoDeSessao(codigo)]);
}

// sair mais rápida que a de um e-mail que existe.

// Um hash qualquer, feito uma vez por processo, só para gastar o tempo de uma conferência.

async function gastarOTempoDeUmaConferencia(senha: string): Promise<void> {
  hashDeFachada ??= hashDaSenha("a senha de fachada da entrada");
  await senhaConfere(senha, await hashDeFachada);
}

export async function conferirEntrada(emailBruto: string, senha: string): Promise<string | null> {
  const email = normalizarEmail(emailBruto);
  const login = email ? await loginParaEntrar(email) : null;
  if (!login) {
    await gastarOTempoDeUmaConferencia(senha);
    return null;
  }

  // A PASSAGEM (seção 6): só um login sem senha aceita a senha única, e só enquanto não tiver senha. O hash
  // é feito ANTES de conferir, para o tempo ser o de uma conferência mesmo quando ela não bate.
  if (login.senha_hash === null) {
    const hash = await hashDaSenha(senha);
    if (login.desativado || !senhaUnicaConfere(senha)) return null;
    return (await gravarSenhaDaPassagem(login.id, hash)) ? login.id : null;
  }

  FRASE_DA_SENHA_NOVA_DEFINIDA,

  FRASE_DO_EMAIL_REPETIDO,
  FRASE_DO_LOGIN_QUE_NAO_EXISTE,

// AS AÇÕES DA PÁGINA LOGINS (docs/specs/2026-11-04-logins.md, seção 4), só para administrador. Cada uma
// começa por `await exigirSessao();` (tests/sessao-nas-actions.test.ts) e, logo depois, confere o
// administrador no servidor: o menu esconder o item não basta, porque Server Action tem endereço próprio.

export type EstadoDoFormulario = { erro?: string; ok?: string };

export async function criarLoginAction(
  _anterior: EstadoDoFormulario,
  formData: FormData
): Promise<EstadoDoFormulario> {
  await exigirSessao();
  await exigirAdministrador();
  const nome = nomeValido(String(formData.get("nome") ?? ""));
  const email = normalizarEmail(String(formData.get("email") ?? ""));
  const senha = String(formData.get("senha") ?? "");
  if (!nome) return { erro: FRASE_DO_NOME };
  if (!email) return { erro: FRASE_DO_EMAIL };
  if (!senhaValida(senha)) return { erro: FRASE_DA_SENHA };
  const r = await criarLogin({ email, nome, senhaHash: await hashDaSenha(senha) });
  if ("erro" in r) return { erro: FRASE_DO_EMAIL_REPETIDO };
  revalidatePath("/logins");
  return { ok: `Login de ${nome} criado. Passe a senha inicial a ${nome}.` };
}

export async function definirSenhaAction(
  _anterior: EstadoDoFormulario,
  formData: FormData
): Promise<EstadoDoFormulario> {
  await exigirSessao();
  await exigirAdministrador();
  const senha = String(formData.get("senha") ?? "");
  if (!senhaValida(senha)) return { erro: FRASE_DA_SENHA };
  const ok = await definirSenha(String(formData.get("id") ?? ""), await hashDaSenha(senha));
  if (!ok) return { erro: FRASE_DO_LOGIN_QUE_NAO_EXISTE };
  revalidatePath("/logins");
  return { ok: FRASE_DA_SENHA_NOVA_DEFINIDA };
}

export async function desativarLoginAction(formData: FormData): Promise<void> {
  await exigirSessao();
  const quem = await exigirAdministrador();
  const resultado = await desativarLogin(String(formData.get("id") ?? ""), quem.id);
  revalidatePath("/logins");
  redirect(`/logins?aviso=${resultado}`);
}

export async function reativarLoginAction(formData: FormData): Promise<void> {
  await exigirSessao();
  await exigirAdministrador();
  const ok = await reativarLogin(String(formData.get("id") ?? ""));
  revalidatePath("/logins");
  redirect(`/logins?aviso=${ok ? "reativado" : "nao_encontrado"}`);

"use server";
import { redirect } from "next/navigation";
import { exigirSessao } from "@/lib/exigir-sessao";
import { ipDoCliente } from "@/lib/ip-do-cliente";
import { hashDaSenha, senhaConfere, senhaValida } from "@/lib/logins";
import { senhaHashDoLogin, trocarPropriaSenha } from "@/lib/logins-repositorio";
import {
  FRASE_DA_CONFIRMACAO,
  FRASE_DA_SENHA,
  FRASE_DA_SENHA_ATUAL,
  FRASE_DA_SENHA_TROCADA,
  FRASE_DO_LOGIN_QUE_NAO_EXISTE,
} from "@/lib/logins-textos";
import { clearAttempts, isLockedOut, LOCKOUT_MESSAGE, recordFailure } from "@/lib/login-throttle";
import { fecharSessao, sessaoAtual } from "@/lib/sessao";

// A MINHA CONTA E O SAIR (docs/specs/2026-11-04-logins.md, seção 5), para qualquer login.

export type EstadoDaTroca = { erro?: string; ok?: string };

/**
 * A TROCA DA PRÓPRIA SENHA. A senha atual passa pelo MESMO FREIO da entrada (achado 171): ela é conferida
 * pelo scrypt, e sem freio uma sessão roubada testaria senhas à vontade e, achando, trocaria a senha e
 * trancaria o dono. A troca derruba as OUTRAS sessões do login e mantém a de quem trocou.
 */
export async function trocarMinhaSenha(_anterior: EstadoDaTroca, formData: FormData): Promise<EstadoDaTroca> {
  await exigirSessao();
  const login = await sessaoAtual();
  if (!login) redirect("/entrar");
  const ip = await ipDoCliente();
  if (await isLockedOut(ip)) return { erro: LOCKOUT_MESSAGE };

  const hash = await senhaHashDoLogin(login.id);
  if (!hash || !(await senhaConfere(String(formData.get("atual") ?? ""), hash))) {
    await recordFailure(ip);
    return { erro: FRASE_DA_SENHA_ATUAL };
  }
  await clearAttempts(ip);

  const nova = String(formData.get("nova") ?? "");
  if (!senhaValida(nova)) return { erro: FRASE_DA_SENHA };
  if (nova !== String(formData.get("confirmacao") ?? "")) return { erro: FRASE_DA_CONFIRMACAO };
  const ok = await trocarPropriaSenha(login.id, await hashDaSenha(nova), login.sessao_id);
  return ok ? { ok: FRASE_DA_SENHA_TROCADA } : { erro: FRASE_DO_LOGIN_QUE_NAO_EXISTE };
}

/**
 * O SAIR: apaga a sessão no banco e o cookie, e volta para a tela de entrar. É uma action, chamada por um
 * formulário (POST), e nunca um GET nem um link (achado 169): o Next pré-carrega os links e as rotas GET, e
 * um Sair por GET derrubaria a sessão de quem só passou o mouse.
 */
export async function sair(): Promise<void> {
  await exigirSessao();
  await fecharSessao();
  redirect("/entrar");
}

  console.error("RECUSADO: sem DATABASE_URL no processo.");
  process.exit(2);
}
const host = new URL(url).hostname;
console.log(`alvo: ${host}`);
const sql = postgres(url, {
  prepare: false,
  ssl: ["127.0.0.1", "localhost"].includes(host) ? false : "require",
  max: 1,
});
try {
  const [tabela] = await sql`select to_regclass('logins') is not null as existe`;
  if (!tabela.existe) throw new Error("RECUSADO: a tabela logins não existe. A 022 foi aplicada?");
  const [{ n }] = await sql`select count(*)::int as n from logins`;
  if (n !== 0) throw new Error(`RECUSADO: já há ${n} login(s), e a passagem só cria o primeiro.`);
  const [linha] = await sql`
    insert into logins (email, nome, senha_hash, administrador) values (${email}, ${"Eduardo"}, null, true)
    returning id`;
  const [conferido] = await sql`
    select count(*)::int as total, bool_and(senha_hash is null) as sem_senha, bool_and(administrador) as administrador
      from logins`;
  console.log(
    `login criado: id ${linha.id}; total ${conferido.total}; senha nula: ${conferido.sem_senha}; ` +
      `administrador: ${conferido.administrador}`
  );
} catch (e) {
  console.error(e instanceof Error ? e.message : String(e));
  process.exitCode = 1;
} finally {
  await sql.end();
}
